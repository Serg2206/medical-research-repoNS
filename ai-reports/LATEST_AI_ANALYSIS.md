5. **Оптимизация использования памяти:** В текущем коде токены переносятся на устройство (GPU или CPU) один за другим. Это может быть неэффективно, особенно если у вас много токенов. Вместо этого, вы можете перенести все токены на устройство одновременно.

Вот как можно было бы улучшить код, учитывая эти рекомендации:

```python
import argparse
import torch
from transformers import AutoModelForSequenceClassification, AutoTokenizer

DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")

def init_model(model_dir):
    """Инициализация модели и токенизатора."""
    tokenizer = AutoTokenizer.from_pretrained(model_dir)
    model = AutoModelForSequenceClassification.from_pretrained(model_dir)
    model.to(DEVICE)
    return tokenizer, model

def infer_model(tokenizer, model, text):
    """Предсказание с помощью модели."""
    tokens = tokenizer(
        text, return_tensors="pt", truncation=True, padding=True, max_length=128
    )
    tokens = {k: v.to(DEVICE) for k, v in tokens.items()}

    model.eval()
    with torch.no_grad():
        outputs = model(**tokens)
        probabilities = torch.softmax(outputs.logits, dim=-1)

    return probabilities.tolist()

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Инференс с медицинской нейросетью")
    parser.add_argument("--model", required=True, help="Путь к сохраненной модели")
    parser.add_argument("--text", required=True, help="Текст для анализа")
    args = parser.parse_args()

    try:
        tokenizer, model = init_model(args.model)
        result = infer_model(tokenizer, model, args.text)
        print(f"Результаты предсказания: {result}")
    except Exception as e:
        print(f"Произошла ошибка: {e}")
```
